---------------------------------------------------------
1. FIRESTORE RULES (Paste this in Firestore Database -> Rules)
---------------------------------------------------------

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if the user is in the whitelist collection by UID
    function isWhitelisted() {
      return exists(/databases/$(database)/documents/whitelist/$(request.auth.uid));
    }

    // Helper to check if the user is an Admin by UID
    function isAdmin() {
      return exists(/databases/$(database)/documents/admin_whitelist/$(request.auth.uid));
    }

    // 1. Whitelist Collection
    // Users can read their own entry to confirm status
    match /whitelist/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false; // Only manual console edits or admin allowed
    }

    // 2. Admin Whitelist Collection
    match /admin_whitelist/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    // 3. User Data & Shared Data
    match /user_data/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    
    match /shared_profile_data/{docId} {
      allow read: if isWhitelisted();
      allow write: if request.auth != null;
    }

    // 4. Content (Gallery, Jokes, Memories)
    // READ: Only Whitelisted users
    // WRITE: Only Admin
    match /{collection}/{document=**} {
      allow read: if isWhitelisted() || isAdmin();
      allow write: if isAdmin();
    }

    // 5. Config
    match /config/{document=**} {
      allow read: if isWhitelisted() || isAdmin();
      allow write: if isAdmin();
    }
  }
}
